cmake_minimum_required(VERSION 3.20)
project(vecdex VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(VECDEX_NATIVE "Tune for the build machine" OFF)
option(VECDEX_SANITIZE "Build with ASan and UBSan" OFF)
option(VECDEX_PYTHON "Build the Python module" OFF)

add_library(vecdex STATIC src/index.cpp)
target_include_directories(vecdex PUBLIC include)
target_compile_options(vecdex PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_options(vecdex PUBLIC -fno-math-errno -fno-trapping-math -ffp-contract=fast)
if(VECDEX_NATIVE)
  target_compile_options(vecdex PUBLIC -march=native)
endif()
if(VECDEX_SANITIZE)
  target_compile_options(vecdex PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vecdex PUBLIC -fsanitize=address,undefined)
endif()

if(SKBUILD OR VECDEX_PYTHON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_vecdex python/bind.cpp)
  target_link_libraries(_vecdex PRIVATE vecdex)
  if(SKBUILD)
    install(TARGETS _vecdex LIBRARY DESTINATION vecdex)
  endif()
endif()

if(NOT SKBUILD)
  include(CTest)
  add_executable(vecdex_tests tests/test_index.cpp)
  target_link_libraries(vecdex_tests PRIVATE vecdex)
  add_test(NAME index COMMAND vecdex_tests)
endif()
